#   Docker Compose (recommended):  DB_HOST=db   REDIS_URL=redis://redis:6379

# 32-byte hex key used to encrypt provider signing secrets at rest.

# Dashboard login credentials

# 64-byte hex secret used to sign JWT tokens.
# Generate with: openssl rand -hex 64

DB_USER=postgres

# Docker Postgres container (creates the initial superuser + database)

# Goose migrations — uses these env vars automatically.
